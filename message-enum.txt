# Commands. These are sent from the client to the server
enum ChatCommand: int
    ROLL = 0
    SAY = 1
    NAME = 2
    SHUFFLE = 3
    CHOOSE = 4
    TABLE = 5

struct NameCommand: array
    type = ChatCommand.NAME
    name = str
struct RollCommand: array
    type = ChatCommand.ROLL
    original = str
    expr = UnrolledDiceExpr
struct SayCommand: array
    type = ChatCommand.SAY
    text = str
struct ShuffleCommand: array
    type = ChatCommand.SHUFFLE
    original = str
    items = [str]
struct ChooseCommand: array
    type = ChatCommand.CHOOSE
    original = str
    items = [str]
struct TableCommand: array
    type = ChatCommand.TABLE
    original = str
    table = str

union Command
    NameCommand
    RollCommand
    SayCommand
    ShuffleCommand
    ChooseCommand
    TableCommand

# Dice stuff
enum DieExprTypes: int
    NUMBER = 0 # 4
    DICE = 1 # 2d6
    GROUP = 2 # (2d6+4)
    UNARY = 3 # -3
    BINARY = 4 # d6*2

enum UNARY: int
    PLUS = 0 # +
    NEG = 1 # -
    NOT = 2 # !

enum BINARY: int
    ADD = 0 # +
    SUB = 1 # -
    MUL = 2 # *
    DIV = 3 # /
    EQ = 4  # =
    NE = 5  # !=
    GT = 6  # >
    GE = 7  # >=
    LT = 8  # <
    LE = 9  # <=

# Server responds with rolled dice.
struct RolledNumber: array
    type = DieExprTypes.NUMBER
    value = int

struct RolledDice: array
    type = DieExprTypes.DICE
    n = int
    base = int
    values = [int]

struct RolledGroup: array
    type = DieExprTypes.GROUP
    value = RolledExpr

struct RolledUnary: array
    type = DieExprTypes.UNARY
    op = UNARY
    value = RolledExpr

struct RolledBinary: array
    type = DieExprTypes.BINARY
    op = BINARY
    left = RolledExpr
    right = RolledExpr

# Use the type field to discriminate
union RolledExpr
    RolledNumber
    RolledDice
    RolledGroup
    RolledUnary
    RolledBinary

# Client sends unrolled dice
struct UnrolledDice: array
    type = DieExprTypes.DICE
    n = int
    base = int

struct UnrolledGroup: array
    type = DieExprTypes.GROUP
    value = UnrolledDiceExpr

struct UnrolledUnary: array
    type = DieExprTypes.UNARY
    op = UNARY
    value = UnrolledDiceExpr

struct UnrolledBinary: array
    type = DieExprTypes.BINARY
    op = BINARY
    left = UnrolledDiceExpr
    right = UnrolledDiceExpr

# Use the type field to discriminate
union UnrolledDiceExpr
    RolledNumber
    UnrolledDice
    UnrolledGroup
    UnrolledUnary
    UnrolledBinary

# Chat messages sent from the server to the client.
# These are in response to Commands
enum Msg: int
    PLAIN = 0
    DICEROLL = 1
    ERROR = 2
    DICEROLL_NEW = 3
    ECHOABLE_TEXT = 4

struct ChatPlainMsg: array
    type = Msg.PLAIN
    user = str
    data = str

struct ChatErrorMsg: array
    type = Msg.ERROR
    user = str
    data = str

struct ChatNewRollMsg: array
    type = Msg.DICEROLL_NEW
    user = str
    data = RolledExpr

struct ChatEchoableMsg: array
    type = Msg.ECHOABLE_TEXT
    user = str
    data = str

# The actual chat message union. Use the type field to discriminate.
union ChatMsg
    ChatPlainMsg
    ChatErrorMsg
    ChatNewRollMsg
    ChatEchoableMsg

# Roll table for the table command
# Tables are stored on the server, clients can create and update them.
enum ROLLTABLEFLAGS: int
    NONE = 0
    IS_TABLE = 1 << 0

struct RollTableEntry: dict
    weight = int
    value = str
    flags = ROLLTABLEFLAGS

struct RollTable: dict
    name = str
    id = int
    entries = [RollTableEntry]
